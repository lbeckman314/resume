\newcommand{\myAdmonition}[3]{%
    \begin{tcolorbox}[
        breakable,
        colback=white,
        colframe=#2,
        leftrule=3pt, toprule=0pt, bottomrule=0pt, rightrule=0pt,
        arc=0pt,
        left=8pt, right=4pt, top=4pt, bottom=4pt,
    ]
    {\small\bfseries\textcolor{#2}{#3}}\\[2pt]
}
\newcommand{\myAdmonitionEnd}{\end{tcolorbox}}

\newenvironment{mynote}{\myAdmonition{adm-note-fr}{adm-note-fr}{\faInfoCircle\ Note}}{\myAdmonitionEnd}
